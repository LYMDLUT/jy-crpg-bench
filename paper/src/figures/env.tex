% row gap 0.62 (label line). Layers: L0 frames and the axes panel, L1 rules,
% L2 panel letters and notes, L3 the axes drawing.

\usepackage{times}
\usepackage[T1]{fontenc}
\usetikzlibrary{arrows.meta,calc}
\graphicspath{{.}{figures/}}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily,
  >={Stealth[length=2.2mm,width=1.6mm]},
  sub/.style={font=\rmfamily\footnotesize,text=ink,inner sep=0,anchor=north west},
  dim/.style={font=\sffamily\scriptsize,text=soft,inner sep=0},
  keyname/.style={font=\sffamily\small\bfseries,text=accent,inner sep=0},
  alias/.style={font=\sffamily\scriptsize,text=soft,inner sep=0},
]
% ---- L0 + L1 + L2: the six frames --------------------------------------
\foreach \col/\row/\file/\letter/\cap in {%
  0/0/room-x3/(a)/{The Scene Every Session Starts In},
  1/0/map-x3/(b)/{The Overworld the Scenes Hang Off},
  0/1/dialogue-x3/(d)/{Dialogue, 16-Pixel Glyphs},

  2/1/sheet-x3/(f)/{The Record the Benchmark Reads}}

  \pgfmathsetmacro{\x}{\col*(\FW+\FG)}
  \pgfmathsetmacro{\y}{-\row*(\FH+\FR)}
  \node[inner sep=0,anchor=north west] (f) at (\x,\y) {\includegraphics[width=\FW cm]{\file}};
  \draw[rule,line width=0.4pt] (f.north west) rectangle (f.south east);
  \node[sub] at ($(f.south west)+(0,-0.12)$) {\letter\ \cap};

% ---- (c) the axes panel, same box as a frame --------------------------
\pgfmathsetmacro{\cx}{2*(\FW+\FG)}
\begin{scope}[shift={(\cx,0)}]
  \fill[tile] (0,0) rectangle (\FW,-\FH);
  \draw[rule,line width=0.4pt] (0,0) rectangle (\FW,-\FH);
  \node[sub] at (0,-\FH-0.12) {(c)\ The Four Axes the Game Moves Along};
  % L3: the isometric tile and its four axes; centre of the panel
  \coordinate (o) at (\FW/2, -\FH/2+0.05);
  % tile diamond, 2:1 isometric
  \fill[wash] ($(o)+(0,0.55)$) -- ($(o)+(1.10,0)$) -- ($(o)+(0,-0.55)$) -- ($(o)+(-1.10,0)$) -- cycle;
  \draw[accent,line width=0.5pt] ($(o)+(0,0.55)$) -- ($(o)+(1.10,0)$) -- ($(o)+(0,-0.55)$) -- ($(o)+(-1.10,0)$) -- cycle;
  \fill[accent] (o) circle (0.07);
  % four movement axes at the 2:1 slope of the projection
  \draw[->,accent,line width=0.8pt] (o) -- ($(o)+(-1.30,0.65)$);
  \draw[->,accent,line width=0.8pt] (o) -- ($(o)+(1.30,0.65)$);
  \draw[->,accent,line width=0.8pt] (o) -- ($(o)+(-1.30,-0.65)$);
  \draw[->,accent,line width=0.8pt] (o) -- ($(o)+(1.30,-0.65)$);
  \node[keyname,anchor=south east] at ($(o)+(-1.36,0.70)$) {kp7};

  \node[keyname,anchor=north east] at ($(o)+(-1.36,-0.70)$) {kp1};
  \node[alias,anchor=north east] at ($(o)+(-1.36,-1.02)$) {down};
  \node[keyname,anchor=north west] at ($(o)+(1.36,-0.70)$) {kp3};
  \node[alias,anchor=north west] at ($(o)+(1.36,-1.02)$) {right};
  % the screen-vertical no key produces
  \draw[soft,line width=0.5pt,dash pattern=on 1.2pt off 1.2pt] ($(o)+(0,0.72)$) -- ($(o)+(0,1.12)$);
  \node[dim,anchor=south] at ($(o)+(0,1.14)$) {No Key Moves Straight Up the Screen};
