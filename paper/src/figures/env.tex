% Figure 1. The game as a state chart of three states, the world map, a
% location and a battle, with the transitions between them and the steps each
% repeats, and the frames the model receives around it, each pointing at the
% state it shows.
% Grid: 1 unit = 1 cm, four columns of W=3.25 at pitch C=3.55 (13.90 wide).
% Rows: top frames (captions above), the band of states, bottom frames
% (captions below). The three state boxes sit under columns 2-4, 2.90 wide,
% with 0.80 gaps that carry the transitions; column 1 of the band holds the
% goal and the inventory frame.
% Rule: a frame arrow ends on a box edge, in space free of text; a transition
% runs straight across a gap with its label above or below it.
% Layers: z0 boxes, z1 frames, z2 connectors, main all type.
\documentclass[tikz,border=1pt]{standalone}
\usepackage{times}
\usepackage[T1]{fontenc}
\usepackage{amssymb}
\usetikzlibrary{arrows.meta,bending,calc}
\graphicspath{{.}{figures/}}

\definecolor{ink}{HTML}{1C1C1E}
\definecolor{accent}{HTML}{2F6FB0}
\definecolor{rule}{HTML}{DCDDE0}
\definecolor{wash}{HTML}{E8F0F8}
\definecolor{box}{HTML}{F1F2F5}
\definecolor{edge}{HTML}{7E8188}
\definecolor{shell}{HTML}{F8F9FA}

\newcommand{\W}{3.25}
\newcommand{\FH}{2.03}
\newcommand{\C}{3.55}

\pgfdeclarelayer{zk}\pgfdeclarelayer{z0}\pgfdeclarelayer{z1}\pgfdeclarelayer{z2}
\pgfsetlayers{zk,z0,z1,z2,main}

\begin{document}
\begin{tikzpicture}[
  >={Stealth[length=1.8mm,width=1.3mm]},
  sub/.style={font=\rmfamily\footnotesize,text=ink,inner sep=0},
  title/.style={font=\sffamily\footnotesize\bfseries,text=ink,inner sep=0,anchor=north west},
  entry/.style={font=\sffamily\scriptsize,text=ink,inner sep=0,anchor=north west,align=left},
  tlabel/.style={font=\sffamily\scriptsize\itshape,text=ink,inner sep=0},
  link/.style={-{Circle[length=1.1mm]},ink,line width=0.5pt},
  trans/.style={->,ink,line width=0.9pt},
  state/.style={draw=edge,fill=box,rounded corners=4pt,line width=0.6pt},
]
% ---------------------------------------------------------------- rows
\def\Ttop{-0.36}   % top of the top frames; captions above
\def\Stop{-2.80}   % top of the state boxes
\def\Sbot{-5.40}   % bottom of the state boxes
\def\Kt{-2.50}     % the container of the three states, top
\def\Kb{-5.92}     % and bottom
\def\Ftop{-6.35}   % top of the bottom frames; captions below
% state boxes: x ranges
\def\Aa{3.55}  \def\Ab{6.41}    % world map
\def\Ba{7.11}  \def\Bb{9.96}    % location
\def\Ca{10.66} \def\Cb{13.52}   % battle; the corner loop needs the last 0.38
\def\P{0.335}                   % line pitch of the entries

% ---------------------------------------------------------------- frames
\foreach \col/\file/\nm/\cap in {%
  0/sheet-x3/ff/{(f) a character sheet},
  1/map-x3/fb/{(b) the world map},
  2/room-x3/fc/{(c) the starting house},
  3/dialogue-x3/fd/{(d) dialogue, 16-pixel glyphs}}
{
  \begin{pgfonlayer}{z1}
  \node[inner sep=0,anchor=north west] (\nm) at (\col*\C,\Ttop) {\includegraphics[width=\W cm]{\file}};
  \draw[rule,line width=0.4pt] (\nm.north west) rectangle (\nm.south east);
  \end{pgfonlayer}
  \node[sub,anchor=south west] at ($(\nm.north west)+(0,0.10)$) {\cap};
}
\foreach \col/\file/\nm/\cap in {%
  1/party-x3/fg/{(g) world-map menu, party},
  2/menu-x3/fh/{(h) the menu in a location},
  3/battle-x3/fi/{(i) a turn of a battle}}
{
  \begin{pgfonlayer}{z1}
  \node[inner sep=0,anchor=north west] (\nm) at (\col*\C,\Ftop) {\includegraphics[width=\W cm]{\file}};
  \draw[rule,line width=0.4pt] (\nm.north west) rectangle (\nm.south east);
  \end{pgfonlayer}
  \node[sub,anchor=north west] at ($(\nm.south west)+(0,-0.10)$) {\cap};
}
% band, column 1: the goal, then (e) the inventory
\node[entry,text width=2.95cm,align=flush left] (goal) at (0,\Stop)
  {\footnotesize\textbf{Goal}\ \ the fourteen books, then the final battle and the ending};
\begin{pgfonlayer}{z1}
\node[inner sep=0,anchor=north west] (fe) at (0,\Stop-0.94) {\includegraphics[width=\W cm]{inventory-x3}};

\end{pgfonlayer}
\node[sub,anchor=north west] at ($(fe.south west)+(0,-0.10)$) {(e) the inventory};

% ---------------------------------------------------------------- (a) axes, bottom row column 1
\begin{scope}[shift={(0,\Ftop)}]
  \node[sub,anchor=north west] at (0,-\FH-0.10) {(a) the four movement axes};
  \coordinate (o) at (\W/2-0.15, -\FH/2);
  \fill[wash] ($(o)+(0,0.40)$) -- ($(o)+(0.80,0)$) -- ($(o)+(0,-0.40)$) -- ($(o)+(-0.80,0)$) -- cycle;
  \draw[accent,line width=0.5pt] ($(o)+(0,0.40)$) -- ($(o)+(0.80,0)$) -- ($(o)+(0,-0.40)$) -- ($(o)+(-0.80,0)$) -- cycle;
  \fill[accent] (o) circle (0.06);
  \foreach \dx/\dy in {-1/1,1/1,-1/-1,1/-1}
    \draw[->,accent,line width=0.7pt] (o) -- ($(o)+(\dx*0.80,\dy*0.40)$);
  \foreach \dx/\dy/\k/\a in {-1/1/kp7/left,1/1/kp9/up,-1/-1/kp1/down,1/-1/kp3/right}{
    \node[font=\sffamily\scriptsize\bfseries,text=ink,inner sep=0,anchor=\ifnum\dx<0 east\else west\fi] at ($(o)+(\dx*0.86,\dy*0.52)$) {\k};
    \node[font=\sffamily\scriptsize,text=ink,inner sep=0,anchor=\ifnum\dx<0 east\else west\fi] at ($(o)+(\dx*0.86,\dy*0.52-0.26)$) {\a};
  }
  \coordinate (atop) at ($(o)+(0,0.40)$);
\end{scope}

% ---------------------------------------------------------------- states

  \draw[state] (\Aa,\Stop) rectangle (\Ab,\Sbot);

  \draw[edge,rounded corners=5pt,line width=0.5pt,dash pattern=on 2pt off 1.5pt] (3.40,\Kt) rectangle (13.88,\Kb);
\end{pgfonlayer}

\newcommand{\statetitle}[3]{% name, x, text
  \node[title] (#1) at (#2+0.12,\Stop-0.14) {#3};}
% one node per line, named <prefix><n>, so a connector can land on a line
\newcommand{\entries}[3]{% prefix, x, lines
  \foreach \t [count=\n] in {#3}
    \node[entry] (#1\n) at (#2+0.12,{\Stop-0.60-(\n-1)*\P}) {\t};}
% a self-loop on the upper-right corner: a circle about the corner, the three
% quarters outside the box, from the top edge round into the right edge
\newcommand{\selfloop}[1]{% right x of the box
  \draw[-{Stealth[length=1.8mm,width=1.3mm,bend]},ink,line width=0.7pt] (#1-0.26,\Stop) arc[start angle=180,end angle=-90,radius=0.26];}
\statetitle{tA}{\Aa}{World map}
\entries{a}{\Aa}{sail the boat, {menu: heal, cure}, dismiss a party member, save and load}
\statetitle{tB}{\Ba}{Location}
\entries{b}{\Ba}{{talk, read}, {search, take items}, use a story item, open a storyline, recruit a party member, collect a book}
\statetitle{tC}{\Ca}{Battle}
\entries{c}{\Ca}{{move, attack with a}, {martial art, poison,}, {cure, heal, items,}, {wait, rest, auto}, {a loss can end it}, {the final battle ends it}}
\selfloop{\Ab}\selfloop{\Bb}\selfloop{\Cb}

% ---------------------------------------------------------------- transitions
\def\Yup{-3.75} \def\Ydn{-4.55}

\draw[trans] (\Bb,\Yup) -- (\Ca,\Yup);
\node[tlabel,anchor=south] at ({(\Bb+\Ca)/2},\Yup+0.07) {event};
\draw[trans] (\Ca,\Ydn) -- (\Bb,\Ydn);
\node[tlabel,anchor=north] at ({(\Bb+\Ca)/2},\Ydn-0.07) {won};
